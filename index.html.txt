<script>
    let rawData = {};
    let filteredQuestions = [];
    let currentQuestionIndex = 0;
    let userAnswers = {};
    let timerInterval = null;
    let timeRemainingSeconds = 0;

    const uploadScreen = document.getElementById('upload-screen');
    const configScreen = document.getElementById('config-screen');
    const examScreen = document.getElementById('exam-screen');
    const resultsScreen = document.getElementById('results-screen');

    const jsonFileInput = document.getElementById('jsonFile');
    const loadJsonBtn = document.getElementById('loadJsonBtn');
    const sectionsContainer = document.getElementById('sectionsContainer');
    const timeLimitInput = document.getElementById('timeLimit');
    const startExamBtn = document.getElementById('startExamBtn');
    
    const timerDisplay = document.getElementById('timer');
    const questionContainer = document.getElementById('questionContainer');
    const prevBtn = document.getElementById('prevBtn');
    const nextBtn = document.getElementById('nextBtn');
    const scoreSummary = document.getElementById('scoreSummary');
    const restartBtn = document.getElementById('restartBtn');

    // Handle File Upload
    jsonFileInput.addEventListener('change', (e) => {
        const file = e.target.files[0];
        if (!file) return;

        const reader = new FileReader();
        reader.onload = function(event) {
            try {
                rawData = JSON.parse(event.target.result);
                loadJsonBtn.classList.remove('hidden');
            } catch (err) {
                alert('Invalid JSON file format.');
                loadJsonBtn.classList.add('hidden');
            }
        };
        reader.readAsText(file);
    });

    // Proceed to Configuration Screen
    loadJsonBtn.addEventListener('click', () => {
        // Extract unique sections
        // Supports structures where questions have a "section" field or are grouped by section objects
        let sections = new Set();
        
        if (Array.isArray(rawData)) {
            rawData.forEach(q => { if (q.section) sections.add(q.section); });
        } else if (rawData.sections && Array.isArray(rawData.sections)) {
            rawData.sections.forEach(s => { if (s.name) sections.add(s.name); });
        } else if (typeof rawData === 'object') {
            // Assume keys are sections if top-level object
            Object.keys(rawData).forEach(key => sections.add(key));
        }

        if (sections.size === 0) {
            // If no explicit sections found, categorize everything under "General"
            sections.add("General");
        }

        sectionsContainer.innerHTML = '';
        sections.forEach(sec => {
            const div = document.createElement('div');
            div.className = 'section-item';
            div.innerHTML = `
                <input type="checkbox" id="sec_${sec}" value="${sec}" checked>
                <label for="sec_${sec}" style="margin-bottom:0; font-weight:normal;">${sec}</label>
            `;
            sectionsContainer.appendChild(div);
        });

        uploadScreen.classList.add('hidden');
        configScreen.classList.remove('hidden');
    });

    // Start Exam Process
    startExamBtn.addEventListener('click', () => {
        const selectedSections = Array.from(document.querySelectorAll('.section-item input:checked'))
            .map(cb => cb.value);

        if (selectedSections.length === 0) {
            alert('Please select at least one section.');
            return;
        }

        const minutes = parseInt(timeLimitInput.value) || 10;
        timeRemainingSeconds = minutes * 60;

        // Gather questions based on selected sections
        filteredQuestions = [];
        if (Array.isArray(rawData)) {
            filteredQuestions = rawData.filter(q => selectedSections.includes(q.section || "General"));
        } else if (rawData.sections && Array.isArray(rawData.sections)) {
            rawData.sections.forEach(s => {
                if (selectedSections.includes(s.name)) {
                    filteredQuestions.push(...s.questions);
                }
            });
        } else {
            selectedSections.forEach(sec => {
                if (rawData[sec] && Array.isArray(rawData[sec])) {
                    filteredQuestions.push(...rawData[sec]);
                }
            });
        }

        if (filteredQuestions.length === 0) {
            alert('No questions found for the selected sections.');
            return;
        }

        configScreen.classList.add('hidden');
        examScreen.classList.remove('hidden');

        currentQuestionIndex = 0;
        userAnswers = {};
        renderQuestion();
        startTimer();
    });

    // Timer Logic
    function startTimer() {
        updateTimerDisplay();
        timerInterval = setInterval(() => {
            timeRemainingSeconds--;
            updateTimerDisplay();
            if (timeRemainingSeconds <= 0) {
                clearInterval(timerInterval);
                finishExam();
            }
        }, 1000);
    }

    function updateTimerDisplay() {
        const mins = Math.floor(timeRemainingSeconds / 60);
        const secs = timeRemainingSeconds % 60;
        timerDisplay.textContent = `Time Remaining: ${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;
    }

    // Render Questions
    function renderQuestion() {
        const q = filteredQuestions[currentQuestionIndex];
        const savedAnswer = userAnswers[currentQuestionIndex];

        let optionsHtml = '';
        if (q.options) {
            q.options.forEach((opt, idx) => {
                const checked = savedAnswer === idx ? 'checked' : '';
                optionsHtml += `
                    <label>
                        <input type="radio" name="question_option" value="${idx}" ${checked}>
                        ${opt}
                    </label>
                `;
            });
        }

        questionContainer.innerHTML = `
            <h3>Question ${currentQuestionIndex + 1} of ${filteredQuestions.length}</h3>
            <p><strong>${q.question || q.text}</strong></p>
            <div class="options-list">${optionsHtml}</div>
        `;

        prevBtn.classList.toggle('hidden', currentQuestionIndex === 0);
        nextBtn.textContent = currentQuestionIndex === filteredQuestions.length - 1 ? 'Finish Exam' : 'Next';
    }

    // Save current answer
    function saveCurrentAnswer() {
        const selectedOption = document.querySelector('input[name="question_option"]:checked');
        if (selectedOption) {
            userAnswers[currentQuestionIndex] = parseInt(selectedOption.value);
        }
    }

    prevBtn.addEventListener('click', () => {
        saveCurrentAnswer();
        if (currentQuestionIndex > 0) {
            currentQuestionIndex--;
            renderQuestion();
        }
    });

    nextBtn.addEventListener('click', () => {
        saveCurrentAnswer();
        if (currentQuestionIndex < filteredQuestions.length - 1) {
            currentQuestionIndex++;
            renderQuestion();
        } else {
            clearInterval(timerInterval);
            finishExam();
        }
    });

    function finishExam() {
        examScreen.classList.add('hidden');
        resultsScreen.classList.remove('hidden');

        let score = 0;
        filteredQuestions.forEach((q, idx) => {
            const correctIndex = q.answer !== undefined ? q.answer : q.correctAnswer;
            if (userAnswers[idx] === correctIndex) {
                score++;
            }
        });

        scoreSummary.textContent = `You scored ${score} out of ${filteredQuestions.length} (${Math.round((score / filteredQuestions.length) * 100)}%)`;
    }

    restartBtn.addEventListener('click', () => {
        resultsScreen.classList.add('hidden');
        uploadScreen.classList.remove('hidden');
        jsonFileInput.value = '';
        loadJsonBtn.classList.add('hidden');
    });
</script>
